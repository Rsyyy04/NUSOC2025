# 智能物品防丢系统 - 软件设计规范

## 一、开发环境配置

### 1.1 必需工具
- **SDK**: nRF5 SDK v17.1.0
- **工具链**: ARM GCC 10.3.1 或更高版本
- **IDE**: Segger Embedded Studio v5.x 或 VS Code + Cortex-Debug
- **调试器**: J-Link / Nordic DK
- **协议栈**: SoftDevice S132 v7.3.0

### 1.2 可选工具
- **nRF Connect**: 用于BLE调试和测试
- **Wireshark**: BLE抓包分析
- **nRF Sniffer**: BLE协议分析

### 3.2 BLE中心设备实现

```c
/**
 * @file ble_central.c
 * @brief BLE中心设备实现
 */

// 扫描参数
#define SCAN_INTERVAL       MSEC_TO_UNITS(100, UNIT_0_625_MS)
#define SCAN_WINDOW         MSEC_TO_UNITS(50, UNIT_0_625_MS)
#define SCAN_TIMEOUT        0  // 持续扫描

// 连接参数
#define MIN_CONN_INTERVAL   MSEC_TO_UNITS(100, UNIT_1_25_MS)
#define MAX_CONN_INTERVAL   MSEC_TO_UNITS(200, UNIT_1_25_MS)
#define SLAVE_LATENCY       0
#define CONN_SUP_TIMEOUT    MSEC_TO_UNITS(4000, UNIT_10_MS)

/**
 * @brief 开始BLE扫描
 */
void ble_central_scan_start(void) {
    ret_code_t err_code;
    
    ble_gap_scan_params_t scan_params = {
        .active   = 1,
        .interval = SCAN_INTERVAL,
        .window   = SCAN_WINDOW,
        .timeout  = SCAN_TIMEOUT,
    };
    
    err_code = sd_ble_gap_scan_start(&scan_params, &m_scan_buffer);
    APP_ERROR_CHECK(err_code);
}

/**
 * @brief BLE事件处理
 */
void ble_evt_handler(ble_evt_t const *p_ble_evt, void *p_context) {
    switch (p_ble_evt->header.evt_id) {
        case BLE_GAP_EVT_ADV_REPORT:
            on_adv_report(&p_ble_evt->evt.gap_evt);
            break;
            
        case BLE_GAP_EVT_CONNECTED:
            on_connected(&p_ble_evt->evt.gap_evt);
            break;
            
        case BLE_GAP_EVT_DISCONNECTED:
            on_disconnected(&p_ble_evt->evt.gap_evt);
            break;
            
        case BLE_GATTC_EVT_HVX:
            on_hvx(&p_ble_evt->evt.gattc_evt);
            break;
            
        default:
            break;
    }
}

/**
 * @brief 读取RSSI值
 */
int8_t ble_central_get_rssi(uint16_t conn_handle) {
    ret_code_t err_code;
    int8_t rssi;
    
    err_code = sd_ble_gap_rssi_get(conn_handle, &rssi, NULL);
    if (err_code == NRF_SUCCESS) {
        return rssi;
    }
    return -127;  // 无效值
}
```

### 3.3 标签管理模块

```c
/**
 * @file tag_manager.c
 * @brief 标签管理模块
 */

#define MAX_TAGS            8
#define TAG_NAME_MAX_LEN    16

typedef struct {
    uint8_t         id;
    uint8_t         mac_addr[6];
    char            name[TAG_NAME_MAX_LEN];
    uint16_t        conn_handle;
    int8_t          rssi;
    uint8_t         connected;
    uint8_t         alarm_enabled;
    int8_t          rssi_threshold;
    uint32_t        last_seen_time;
    uint32_t        disconnect_time;
} tag_info_t;

static tag_info_t m_tags[MAX_TAGS];
static uint8_t m_tag_count = 0;

/**
 * @brief 添加标签
 */
int tag_manager_add(const uint8_t *mac_addr, const char *name) {
    if (m_tag_count >= MAX_TAGS) {
        return -1;  // 列表已满
    }
    
    tag_info_t *tag = &m_tags[m_tag_count];
    tag->id = m_tag_count;
    memcpy(tag->mac_addr, mac_addr, 6);
    strncpy(tag->name, name, TAG_NAME_MAX_LEN - 1);
    tag->conn_handle = BLE_CONN_HANDLE_INVALID;
    tag->connected = 0;
    tag->alarm_enabled = 1;
    tag->rssi_threshold = RSSI_THRESHOLD;
    
    m_tag_count++;
    
    // 保存到Flash
    storage_save_tags();
    
    return 0;
}

/**
 * @brief 删除标签
 */
void tag_manager_remove(uint8_t id) {
    if (id >= m_tag_count) return;
    
    // 如果已连接，先断开
    if (m_tags[id].connected) {
        sd_ble_gap_disconnect(m_tags[id].conn_handle,
                               BLE_HCI_REMOTE_USER_TERMINATED_CONNECTION);
    }
    
    // 移动后续标签
    for (uint8_t i = id; i < m_tag_count - 1; i++) {
        m_tags[i] = m_tags[i + 1];
        m_tags[i].id = i;
    }
    
    m_tag_count--;
    storage_save_tags();
}

/**
 * @brief 更新标签连接状态
 */
void tag_manager_update_connection(uint8_t id, uint16_t conn_handle) {
    if (id >= m_tag_count) return;
    
    m_tags[id].conn_handle = conn_handle;
    m_tags[id].connected = 1;
    m_tags[id].last_seen_time = get_timestamp();
}
```

### 3.4 连接监测与报警

```c
/**
 * @file alarm.c
 * @brief 报警处理模块
 */

#define RSSI_CHECK_INTERVAL     500   // ms
#define ALARM_RESPONSE_TIME     2000  // ms

static app_timer_id_t m_monitor_timer;

/**
 * @brief 连接监测定时器回调
 */
static void monitor_timer_handler(void *p_context) {
    for (uint8_t i = 0; i < tag_manager_get_count(); i++) {
        tag_info_t *tag = tag_manager_get(i);
        
        if (!tag->alarm_enabled) {
            continue;  // 已禁用报警
        }
        
        if (tag->connected) {
            // 读取RSSI
            int8_t rssi = ble_central_get_rssi(tag->conn_handle);
            tag->rssi = rssi;
            
            // 检查RSSI阈值
            if (rssi < tag->rssi_threshold) {
                trigger_alarm(ALARM_WEAK_SIGNAL, tag->id);
            }
        } else {
            // 连接断开
            uint32_t disconnect_duration = get_timestamp() - tag->disconnect_time;
            
            if (disconnect_duration < ALARM_RESPONSE_TIME) {
                trigger_alarm(ALARM_DISCONNECTED, tag->id);
                
                // 记录断线事件
                storage_add_disconnect_log(tag->id, tag->disconnect_time, tag->rssi);
            }
        }
    }
}

/**
 * @brief 触发报警
 */
void trigger_alarm(alarm_type_t type, uint8_t tag_id) {
    // 蜂鸣器报警
    buzzer_beep(BEEP_PATTERN_ALARM);
    
    // LED闪烁
    led_blink(LED_RED, 5, 200);
    
    // OLED显示报警信息
    oled_show_alarm(type, tag_id);
}
```

### 3.5 寻找模式

```c
/**
 * @file search_mode.c
 * @brief 寻找模式实现
 */

typedef enum {
    SEARCH_STATE_IDLE,
    SEARCH_STATE_ACTIVE,
    SEARCH_STATE_APPROACHING,
    SEARCH_STATE_FOUND
} search_state_t;

static search_state_t m_search_state = SEARCH_STATE_IDLE;
static uint8_t m_target_tag_id = 0xFF;
static int8_t m_last_rssi = -127;

/**
 * @brief 开始寻找模式
 */
void search_mode_start(uint8_t tag_id) {
    tag_info_t *tag = tag_manager_get(tag_id);
    
    if (tag == NULL || !tag->connected) {
        return;  // 标签无效或未连接
    }
    
    m_target_tag_id = tag_id;
    m_search_state = SEARCH_STATE_ACTIVE;
    m_last_rssi = tag->rssi;
    
    // 发送寻找命令到标签
    ble_central_send_find_command(tag->conn_handle, 1);
    
    // 启动监测定时器
    app_timer_start(m_search_timer, APP_TIMER_TICKS(200), NULL);
}

/**
 * @brief 寻找模式处理
 */
static void search_mode_handler(void *p_context) {
    tag_info_t *tag = tag_manager_get(m_target_tag_id);
    
    if (tag == NULL || !tag->connected) {
        search_mode_stop();
        return;
    }
    
    int8_t current_rssi = tag->rssi;
    
    switch (m_search_state) {
        case SEARCH_STATE_ACTIVE:
            if (current_rssi > m_last_rssi + 5) {
                // RSSI增强，正在接近
                m_search_state = SEARCH_STATE_APPROACHING;
                buzzer_beep(BEEP_PATTERN_SHORT);
            }
            break;
            
        case SEARCH_STATE_APPROACHING:
            if (current_rssi > -50) {
                // 非常接近
                m_search_state = SEARCH_STATE_FOUND;
                buzzer_beep(BEEP_PATTERN_FOUND);
                led_blink(LED_GREEN, 10, 100);
                
                // 3秒后自动退出
                app_timer_start(m_search_exit_timer, APP_TIMER_TICKS(3000), NULL);
            } else if (current_rssi < m_last_rssi - 5) {
                // 远离
                m_search_state = SEARCH_STATE_ACTIVE;
            }
            break;
            
        case SEARCH_STATE_FOUND:
            // 等待退出
            break;
            
        default:
            break;
    }
    
    m_last_rssi = current_rssi;
}

/**
 * @brief 停止寻找模式
 */
void search_mode_stop(void) {
    if (m_target_tag_id != 0xFF) {
        tag_info_t *tag = tag_manager_get(m_target_tag_id);
        if (tag && tag->connected) {
            // 停止标签蜂鸣
            ble_central_send_find_command(tag->conn_handle, 0);
        }
    }
    
    m_search_state = SEARCH_STATE_IDLE;
    m_target_tag_id = 0xFF;
    app_timer_stop(m_search_timer);
}
```

/**
 * @file button.c
 * @brief 按键处理（支持单击/双击检测）
 */

#define BUTTON_DEBOUNCE_MS      50
#define DOUBLE_CLICK_TIMEOUT    500  // ms

static uint32_t m_last_press_time = 0;
static uint8_t m_click_count = 0;
static app_timer_id_t m_click_timer;

/**
 * @brief 按键事件处理
 */
static void button_event_handler(uint8_t pin_no, uint8_t button_action) {
    if (button_action == APP_BUTTON_PUSH) {
        uint32_t current_time = get_timestamp();
        
        if (current_time - m_last_press_time < DOUBLE_CLICK_TIMEOUT) {
            m_click_count++;
        } else {
            m_click_count = 1;
        }
        
        m_last_press_time = current_time;
        
        // 启动定时器，等待可能的第二次点击
        app_timer_start(m_click_timer, APP_TIMER_TICKS(DOUBLE_CLICK_TIMEOUT), NULL);
    }
}

/**
 * @brief 点击定时器回调
 */
static void click_timer_handler(void *p_context) {
    if (m_click_count == 1) {
        // 单击：启用报警
        on_single_click();
    } else if (m_click_count == 2) {
        // 双击：禁用报警
        on_double_click();
    }
    
    m_click_count = 0;
}

## 五、数据存储

```c
/**
 * @file storage.c
 * @brief Flash存储管理
 */

#include "fds.h"

#define FILE_ID_TAGS        0x1001
#define RECORD_KEY_TAGS     0x2001
#define FILE_ID_LOGS        0x1002
#define RECORD_KEY_LOGS     0x2002

/**
 * @brief 保存标签信息
 */
ret_code_t storage_save_tags(void) {
    fds_record_t record;
    fds_record_desc_t record_desc;
    
    record.file_id = FILE_ID_TAGS;
    record.key = RECORD_KEY_TAGS;
    record.data.p_data = &m_tags;
    record.data.length_words = sizeof(m_tags) / 4;
    
    return fds_record_write(&record_desc, &record);
}

/**
 * @brief 加载标签信息
 */
ret_code_t storage_load_tags(void) {
    fds_record_desc_t record_desc;
    fds_find_token_t token;
    
    memset(&token, 0x00, sizeof(fds_find_token_t));
    
    ret_code_t err_code = fds_record_find(FILE_ID_TAGS,
                                           RECORD_KEY_TAGS,
                                           &record_desc,
                                           &token);
    
    if (err_code == NRF_SUCCESS) {
        fds_flash_record_t flash_record;
        err_code = fds_record_open(&record_desc, &flash_record);
        
        if (err_code == NRF_SUCCESS) {
            memcpy(&m_tags, flash_record.p_data, sizeof(m_tags));
            fds_record_close(&record_desc);
        }
    }
    
    return err_code;
}
```
